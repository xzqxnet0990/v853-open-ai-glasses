cmake_minimum_required(VERSION 3.10)
project(AIGlassesDisplay_V853)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

# V853 SDK路径配置
set(V853_SDK_PATH "${CMAKE_SOURCE_DIR}/../../tina-v853-open" CACHE PATH "V853 SDK Path")
set(V853_BUILDROOT "${V853_SDK_PATH}/buildroot/buildroot-201902" CACHE PATH "V853 Buildroot Path")

# V853交叉编译工具链 (使用toolchain-v853.cmake)
# 在命令行传入: cmake .. -DCMAKE_TOOLCHAIN_FILE=../toolchain-v853.cmake

# 查找Qt5 (V853 Buildroot编译的Qt5)
set(Qt5_DIR "${V853_BUILDROOT}/output/host/usr/lib/cmake/Qt5" CACHE PATH "Qt5 CMake Path")
find_package(Qt5 REQUIRED COMPONENTS Core Widgets Gui)

# AI Glass SDK适配层路径
set(SDK_PATH "${CMAKE_SOURCE_DIR}/../sdk_adapter")

# 包含目录
include_directories(
    ${CMAKE_SOURCE_DIR}
    ${CMAKE_SOURCE_DIR}/widgets
    ${SDK_PATH}/include
    ${Qt5Core_INCLUDE_DIRS}
    ${Qt5Widgets_INCLUDE_DIRS}
    ${Qt5Gui_INCLUDE_DIRS}
)

# 源文件
set(SOURCES
    main.cpp
    mainwindow.cpp
    widgets/battery_widget.cpp
    widgets/camera_widget.cpp
    widgets/settings_widget.cpp
    widgets/base_widget.cpp
)

# 头文件
set(HEADERS
    mainwindow.h
    widgets/battery_widget.h
    widgets/camera_widget.h
    widgets/settings_widget.h
    widgets/base_widget.h
)

# 资源文件
set(RESOURCES
    resources/resources.qrc
)

# 创建可执行文件
add_executable(${PROJECT_NAME} ${SOURCES} ${HEADERS} ${RESOURCES})

# 链接库
target_link_libraries(${PROJECT_NAME}
    Qt5::Core
    Qt5::Widgets
    Qt5::Gui
    ${SDK_PATH}/lib/libai_glass_sdk_v853.a  # 静态链接SDK适配层
    pthread
)

# 编译选项
target_compile_options(${PROJECT_NAME} PRIVATE
    -Wall
    -Wextra
    -O2
)

# 安装规则
install(TARGETS ${PROJECT_NAME}
        RUNTIME DESTINATION /usr/bin)

# V853平台特定配置
message(STATUS "Building for V853 Platform")
message(STATUS "V853 SDK Path: ${V853_SDK_PATH}")
message(STATUS "Qt5 Path: ${Qt5_DIR}")
message(STATUS "SDK Adapter Path: ${SDK_PATH}")
